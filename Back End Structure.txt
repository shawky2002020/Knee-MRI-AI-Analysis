/backend
│
├── /config                 # Configuration files (DB, server settings, etc.)
│   ├── db.js               # Database connection
│   ├── server.js           # Server and app setup (Express)
│   └── authConfig.js       # JWT and Auth setup
│
├── /controllers            # Controllers for each feature (API logic)
│   ├── authController.js   # User registration, login, password recovery
│   ├── mriController.js    # Handle MRI file uploads and AI analysis
│   ├── reportController.js # Handle report generation, download
│   └── adminController.js  # Admin user management, logging
│
├── /models                 # Mongoose models (MongoDB schema)
│   ├── User.js             # User schema (roles, credentials)
│   ├── MriScan.js          # MRI scan schema (file data, metadata)
│   └── Report.js           # Report schema (diagnostic data)
│
├── /routes                 # API routes
│   ├── authRoutes.js       # Routes for authentication (login, signup)
│   ├── mriRoutes.js        # Routes for uploading MRI scans
│   ├── reportRoutes.js     # Routes for retrieving reports
│   └── adminRoutes.js      # Routes for admin features
│
├── /middleware             # Middlewares (e.g., JWT authentication, file validation)
│   ├── authMiddleware.js   # JWT validation, role-based access control (RBAC)
│   ├── uploadMiddleware.js # Handle file upload and validation (size, type)
│   └── errorMiddleware.js  # Centralized error handling
│
├── /services               # Business logic (separates pure logic from controllers)
│   ├── aiService.js        # AI analysis logic (interface with Python API or library)
│   ├── emailService.js     # Send email notifications for reports, registration
│   └── fileService.js      # Logic for file handling (upload, download, storage)
│
├── /utils                  # Utility functions
│   └── helpers.js          # Helper functions for data processing
│
├── /public                 # Static files (optional, for frontend assets)
│
├── server.js               # Main entry point for Node.js/Express app
└── package.json            # NPM dependencies
